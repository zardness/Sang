1) ollama pull qwen2.5-coder:1.5b
2) C:\Users\space\.continue/config.yaml을 아래와 같이

# Continue 설정 파일
name: Local Assistant      # 설정 이름 (아무거나)
version: 1.0.0
schema: v1

models:
  # Tab 자동완성 전용 모델
  - name: Qwen2.5-Coder 1.5B
    provider: ollama              # 로컬 Ollama 사용
    model: qwen2.5-coder:1.5b     # 위에서 pull 받은 모델 이름과 동일해야 함
    roles:
      - autocomplete              # 자동완성 역할로 지정
